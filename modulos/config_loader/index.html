    Módulo: config_loader.py¶

    Carregador de configuração centralizada a partir de config.yaml.

    Funções Principais¶

    def load_config(config_path: Optional[Path] = None) -> Dict[str, Any]
def get_config(key: str, default: Any = None) -> Any
def get_threshold(name: str) -> float
def get_trading_config(name: str) -> Any
def get_okx_config(name: str) -> Any
def get_logging_config() -> Dict[str, Any]
def get_reporting_config() -> Dict[str, Any]
def get_knowledge_config() -> Dict[str, Any]
def get_session_categories() -> Dict[str, list]
def get_market_hours() -> tuple
def is_market_hours(hour: Optional[int] = None) -> bool
def reload_config() -> Dict[str, Any]

    Uso¶

    from config_loader import get_config, get_threshold, get_market_hours

# Notação de ponto
win_rate = get_config('thresholds.win_rate_low')  # 40
risk = get_config('trading.default_risk_percent')  # 1.0

# Helpers específicos
threshold = get_threshold('win_rate_low')  # 40
open_h, close_h = get_market_hours()  # (9, 17)
is_open = is_market_hours(10)  # True
is_open = is_market_hours(20)  # False

# Recarregar (útil para hot-reload)
reload_config()

    Configuração Padrão (get_default_config())¶

    Retorna dict com todos os valores padrão caso config.yaml não exista.
